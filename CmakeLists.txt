cmake_minimun_required(VERSION 3.20)
project(GranularSynth LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    add_compile_options(-Wall -Wextra -Wpedantic -O3)
endif()

include(FetchContent)
FetchContent_Declare(
    clap
    GIT_REPOSITORY https://github.com/free-audio/clap.git
    GIT_TAG main
)
FetchContent_MakeAvailable(clap)

add_library(granular_synth MODULE src/plugin.c)

target_link_libraries(granular_synth PRIVATE clap)
target_include_directories(granular_synth PRIVATE src)

set_target_properties(granular_synth PROPERTIES
    PREFIX ""
    SUFFIX ".clap"
)

if(APPLE)
    set_target_properties(granular_synth PROPERTIES BUNDLE TRUE)
endif()
