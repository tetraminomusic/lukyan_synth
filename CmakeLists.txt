cmake_minimum_required(VERSION 3.20)
project(GranularSynth LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    add_compile_options(-Wall -Wextra -Wpedantic -O3)
endif()

include(FetchContent)
FetchContent_Declare(
    clap
    GIT_REPOSITORY https://github.com/free-audio/clap.git
    GIT_TAG main
)
FetchContent_MakeAvailable(clap)

add_library(granular_synth MODULE 
    src/plugin.c 
    src/gui.c
    src/gui_draw.c
    src/ports.c
    src/events.c
    src/parameters.c
    src/granular.c
    src/oscillator.c
    src/adsr.c
    src/fx.c
)

target_link_libraries(granular_synth PRIVATE clap)
target_include_directories(granular_synth PRIVATE include src)

if(APPLE)
    target_link_libraries(granular_synth PRIVATE "-framework Cocoa" "-framework CoreGraphics")
elseif(WIN32)
    target_link_libraries(granular_synth PRIVATE gdi32 user32)
endif()

set_target_properties(granular_synth PROPERTIES
    OUTPUT_NAME "Lukyan Synth"
    PREFIX ""
    SUFFIX ".clap"
)

if(APPLE)
    set_target_properties(granular_synth PROPERTIES 
        BUNDLE TRUE
        BUNDLE_EXTENSION "clap"
    )

    add_custom_command(TARGET granular_synth POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E make_directory "$ENV{HOME}/Library/Audio/Plug-Ins/CLAP"
        COMMAND ${CMAKE_COMMAND} -E copy_directory
            "$<TARGET_BUNDLE_DIR:granular_synth>"
            "$ENV{HOME}/Library/Audio/Plug-Ins/CLAP/Lukyan Synth.clap"
        COMMENT "Плагин скопирован как Lukyan Synth.clap"
    )
endif()
