cmake_minimum_required(VERSION 3.20)
project(GranularSynth LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    add_compile_options(-Wall -Wextra -Wpedantic -O3)
endif()

include(FetchContent)
FetchContent_Declare(clap GIT_REPOSITORY https://github.com/free-audio/clap.git GIT_TAG main)
FetchContent_MakeAvailable(clap)

if(CMAKE_SYSTEM_PROCESSOR MATCHES "(x86_64|AMD64|amd64)")
    enable_language(ASM)
    set(SIMD_SOURCE src/dsp/simd_math_x86.S)
else()
    set(SIMD_SOURCE src/dsp/simd_math_c.c)
endif()

add_library(granular_synth MODULE 
    src/plugin.c
    src/dsp/granular.c
    src/dsp/oscillator.c
    src/dsp/adsr.c
    src/dsp/fx.c
    src/dsp/parameters.c
    src/dsp/fast_math.c
    src/dsp/arena.c 
    src/gui/gui.c
    src/gui/gui_draw.c
    src/platform/ports.c
    src/platform/events.c
    src/platform/presets.c
    ${SIMD_SOURCE}
)

target_link_libraries(granular_synth PRIVATE clap)
target_include_directories(granular_synth PRIVATE 
    include 
    src 
    src/dsp 
    src/gui 
    src/platform
)

set_target_properties(granular_synth PROPERTIES OUTPUT_NAME "Lukyan Synth" PREFIX "" SUFFIX ".clap")

if(APPLE)
    target_link_libraries(granular_synth PRIVATE "-framework Cocoa" "-framework CoreGraphics")
    set_target_properties(granular_synth PROPERTIES BUNDLE TRUE BUNDLE_EXTENSION "clap")
    add_custom_command(TARGET granular_synth POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E make_directory "$ENV{HOME}/Library/Audio/Plug-Ins/CLAP"
        COMMAND ${CMAKE_COMMAND} -E copy_directory "$<TARGET_BUNDLE_DIR:granular_synth>" "$ENV{HOME}/Library/Audio/Plug-Ins/CLAP/Lukyan Synth.clap")
endif()



# Тесты


enable_testing()

add_executable(test_arena tests/test_arena.c src/dsp/arena.c)
target_include_directories(test_arena PRIVATE include)
add_test(NAME Unit_ArenaAllocator COMMAND test_arena)

add_executable(test_dsp_math tests/test_dsp_math.c src/dsp/fast_math.c ${SIMD_SOURCE})
target_include_directories(test_dsp_math PRIVATE include)
add_test(NAME Unit_DspMathParity COMMAND test_dsp_math)

add_executable(test_presets 
    tests/test_presets.c 
    src/dsp/parameters.c 
    src/platform/presets.c
    src/dsp/granular.c
    src/dsp/oscillator.c
    src/dsp/adsr.c
    src/dsp/fx.c
    src/dsp/fast_math.c
    ${SIMD_SOURCE}
)
target_include_directories(test_presets PRIVATE include src)
target_link_libraries(test_presets PRIVATE clap)
if(WIN32)
    target_link_libraries(test_presets PRIVATE gdi32 user32)
endif()
add_test(NAME Unit_PresetSerialization COMMAND test_presets)


add_executable(test_golden 
    tests/test_golden.c
    src/dsp/parameters.c
    src/dsp/granular.c
    src/dsp/oscillator.c
    src/dsp/adsr.c
    src/dsp/fx.c
    src/dsp/fast_math.c
    ${SIMD_SOURCE}
)
target_include_directories(test_golden PRIVATE include src)
target_link_libraries(test_golden PRIVATE clap)
if(WIN32)
    target_link_libraries(test_golden PRIVATE gdi32 user32)
endif()
add_test(NAME Integration_GoldenAudio COMMAND test_golden)
